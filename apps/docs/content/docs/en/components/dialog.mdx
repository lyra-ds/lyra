# Dialog

Dialog is a controlled modal with portal rendering, focus trapping, scroll locking, and accessible
escape routes built in.

## Usage

<CodeBlock>{`'use client';

import { Button, Dialog } from '@lyra-ds/react';
import { useState } from 'react';

export function Example() {
  const [open, setOpen] = useState(false);

return (

<>
<Button onClick={() => setOpen(true)}>Open dialog</Button>
<Dialog open={open} onClose={() => setOpen(false)} title="Delete project">
This action cannot be undone.
</Dialog>
</>
);
}`}</CodeBlock>
